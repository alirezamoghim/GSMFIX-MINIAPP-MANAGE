        function getOrderFinance(order) {
            const total = Number(order.cost) || 0;
            let status = order.paymentStatus;
            let paid = Number(order.paidAmount);

            if (!status) {
                status = total > 0 ? 'بدهکار' : 'تسویه کامل';
            }
            if (isNaN(paid)) {
                paid = status === 'تسویه کامل' ? total : 0;
            }
            if (status === 'تسویه کامل') {
                paid = total;
            }
            const remaining = Math.max(0, total - paid);
            if (total > 0 && remaining === 0 && status !== 'تسویه کامل') {
                status = 'تسویه کامل';
            }
            return { total, paid, remaining, status };
        }
